// 🌟 双重保险的精确写入逻辑 async exportWithTemplateAndData() { try { let response = await fetch('template.xlsx'); if (!response.ok) throw new Error("未找到 template.xlsx 文件，请确认已上传到 GitHub 仓库根目录。"); let arrayBuffer = await response.arrayBuffer(); let workbook = new ExcelJS.Workbook(); await workbook.xlsx.load(arrayBuffer); // 双重保险：优先通过名称查找，如果找不到直接取第一个 Tab（索引 0）作为 Master Listing let worksheet = workbook.worksheets.find(ws => { let name = ws.name.toLowerCase(); return name.includes('master') || name.includes('listing') || name.includes('明细'); }) || workbook.worksheets[0]; console.log("当前正在向以下工作表写入数据:", worksheet.name); // 清理旧数据行（从第 5 行到第 150 行） for (let r = 5; r <= 150; r++) { let row = worksheet.getRow(r); ['A','B','C','D','E','F','G','H','I','J','K','L'].forEach(col => { row.getCell(col).value = null; }); } // 写入当前公司最新的排序开支数据 let itemsToWrite = this.sortedExpenses; itemsToWrite.forEach((item, index) => { let rowNum = 5 + index; let row = worksheet.getRow(rowNum); row.getCell('A').value = item.month || ''; row.getCell('B').value = item.building || ''; row.getCell('C').value = item.unit || ''; row.getCell('D').value = item.invoiceDate || '待核实'; row.getCell('E').value = item.subDate || ''; row.getCell('F').value = item.desc || ''; row.getCell('G').value = item.payable || ''; row.getCell('H').value = item.invoiceNo || ''; row.getCell('I').value = item.payMode || ''; row.getCell('J').value = item.category || ''; row.getCell('K').value = item.amount !== '' ? Number(item.amount) : '待核实'; row.getCell('L').value = item.remark || ''; }); // 导出包含三个完整 Tab 的 Excel 文件 let buffer = await workbook.xlsx.writeBuffer(); let blob = new Blob([buffer], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' }); saveAs(blob, `${this.currentCompany}_Admin_Expenses_Executive_Hub.xlsx`); } catch (err) { alert("导出失败: " + err.message); } }
